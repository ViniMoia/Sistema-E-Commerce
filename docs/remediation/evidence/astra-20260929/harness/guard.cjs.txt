
const net=require('node:net'),https=require('node:https'),http=require('node:http');
const original=net.Socket.prototype.connect;
function local(host,port){return ['127.0.0.1','localhost','::1'].includes(host) && ![5432,5433].includes(Number(port));}
net.Socket.prototype.connect=function(...args){let o=args[0];if(Array.isArray(o))o=o[0];const host=typeof o==='object'?o.host:args[1];const port=typeof o==='object'?o.port:o;if(local(host||'localhost',port))return original.apply(this,args);throw new Error('ISOLATED_NETWORK_BLOCKED');};
https.request=()=>{throw new Error('ISOLATED_HTTPS_BLOCKED')};https.get=https.request;
const fetch=globalThis.fetch;globalThis.fetch=async (input,options)=>{const u=new URL(typeof input==='string'?input:input.url||input);if(u.protocol==='http:'&&local(u.hostname,u.port))return fetch(input,options);throw new Error('ISOLATED_FETCH_BLOCKED');};
